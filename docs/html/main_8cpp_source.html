126void loop() {
127 //Read sensor inputs
128 apps5V = analogRead(APPS_5V_PIN);
129 apps3V3 = analogRead(APPS_3V3_PIN);
130 brake = analogRead(BRAKE_PIN);
131 startButtonPressed = (digitalRead(START_BUTTON_PIN) == LOW);
132
133 // Update brake light based on threshold
134 digitalWrite(BRAKE_LIGHT_PIN, (brake > BRAKE_DEPRESSED_THRESHOLD) ? HIGH : LOW);
135
136 // State machine handling
137 switch (currentState) {
138 case INIT:
139 // No torque, LEDS/buzzer off
140 calculatedTorque = 0;
141 digitalWrite(DRIVE_LED_PIN, LOW);
142 digitalWrite(BUZZER_PIN, LOW);
143
144 // transition to STARTIN if button pressed and brake depressed
145 if (startButtonPressed && brake > BRAKE_DEPRESSED_THRESHOLD)
146 {
147 currentState = STARTIN;
148 stateStartTime = millis();
149 }
150 break;
151
152 case STARTIN:
153 // No torque, buzzer/LED off
154 calculatedTorque = 0;
155 digitalWrite(BUZZER_PIN, LOW);
156 digitalWrite(DRIVE_LED_PIN, LOW);
157
158 // Check hold time and BMS ready message
159 if (!startButtonPressed || brake <= BRAKE_DEPRESSED_THRESHOLD) {
160 currentState = INIT;
161 break;
162 }
163
164 //Check hold time and BMS ready message
165 if (millis() - stateStartTime >= STARTIN_HOLD_TIME) {
166 can_frame msg;
167 if (canBMS.readMessage(&msg) == MCP2515::ERROR_OK &&
168 msg.can_id == BMS_READY_ID && msg.data[BMS_READY_BYTE] == BMS_READY_VALUE) {
169 currentState = BUZZIN;
170 stateStartTime = millis();
171 }
172 }
173 break;
174
175 case BUZZIN:
176 // No torque, buzzer on, LED off
177 calculatedTorque = 0;
178 digitalWrite(BUZZER_PIN, HIGH);
179 digitalWrite(DRIVE_LED_PIN, LOW);
180
181 // Transition to DRIVE after buzz time
182 if (millis() - stateStartTime >= BUZZIN_TIME) {
183 currentState = DRIVE;
184 stateStartTime = millis();
185 digitalWrite(BUZZER_PIN, LOW);
186 }
187 break;
188
189 case DRIVE:
190 // LED on
191 digitalWrite(DRIVE_LED_PIN, HIGH);
192
193 // Check APPS fault
194 int32_t left = (int32_t)apps5V * 33;
195 int32_t right = (int32_t)apps3V3 *50;
196 int32_t abs_diff_scaled = labs(left-right);
197 int32_t range = PEDAL_MAX - PEDAL_MIN;
198 if (10 * abs_diff_scaled > range * 33) {
199 if (!isFaulty) {
200 faultStartTime = millis();
201 isFaulty = true;
202 }
203 if (millis() - faultStartTime > APPS_FAULT_TIMEOUT) {
204 currentState = INIT;
205 calculatedTorque = 0;
206 isFaulty = false;
207 break;
208 }
209 }
210 else {
211 isFaulty = false;
212 }
213
214 //Calculate Torque
215 int64_t num = static_cast<int64_t>(left) + right - static_cast<int64_t>(PEDAL_MIN) *66;
216 int64_t den = 66LL * static_cast<int64_t>(PEDAL_MAX-PEDAL_MIN);
217 int64_t clamped_num = (num<0) ? 0LL : (num>den)?den:num;
218 int16_t torque_scale = FLIP_MOTOR_DIRECTION ? -32768 : 32767;
219 calculatedTorque = static_cast<int16_t>((clamped_num * static_cast<int64_t>(torque_scale))/den);
220
221 // Send motor torque
222 can_frame msg;
223 msg.can_id = MOTOR_TORQUE_ID;
224 msg.can_dlc = 3;
225 msg.data[0] = 0x90;
226 msg.data[1] = calculatedTorque & 0xFF;
227 msg.data[2] = (calculatedTorque >> 8) & 0xFF;
228 canMotor.sendMessage(&msg);
229 break;
230 }
231
232 // Send debug pedal readings CAN message
233 can_frame pedalMsg;
234 pedalMsg.can_id = DEBUG_PEDALS_ID;
235 pedalMsg.can_dlc = 6;
236 pedalMsg.data[0] = (uint8_t)(apps5V & 0xFF);
237 pedalMsg.data[1] = (uint8_t)(apps5V >> 8);
238 pedalMsg.data[2] = (uint8_t)(apps3V3 & 0xFF);
239 pedalMsg.data[3] = (uint8_t)(apps3V3 >> 8);
240 pedalMsg.data[4] = (uint8_t)(brake & 0xFF);
241 pedalMsg.data[5] = (uint8_t)(brake >> 8);
242 canDebug.sendMessage(&pedalMsg);
243
244 // Send debug fault message if faulty
245 can_frame stateMsg;
246 stateMsg.can_id = DEBUG_STATE_ID;
247 stateMsg.can_dlc = 1;
248 stateMsg.data[0] = (uint8_t)currentState;
249 canDebug.sendMessage(&stateMsg);
250
251 if (isFaulty) {
252 can_frame faultMsg;
253 faultMsg.can_id = DEBUG_FAULT_ID;
254 faultMsg.can_dlc = 8;
255 int32_t left = (int32_t)apps5V * 33;
256 int32_t right = (int32_t)apps3V3 * 50;
257 int32_t diff_scaled = labs(left-right);
258 uint16_t diff = static_cast<int16_t>((diff_scaled + 16) / 33);
259 faultMsg.data[0] = (uint8_t)(diff & 0xFF);
260 faultMsg.data[1] = (uint8_t)(diff >> 8);
261 canDebug.sendMessage(&faultMsg);
262 }
